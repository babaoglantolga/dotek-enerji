create table departmanlar (
  id bigint generated always as identity primary key,
  ad text not null unique,
  ofis boolean not null default false,   -- fatura işleri bu departmana düşer
  olusturma timestamptz default now()
);

create table profiller (
  id uuid primary key references auth.users on delete cascade,
  ad text not null,
  kullanici_adi text,
  rol text not null default 'bekliyor'
    check (rol in ('patron','ds','calisan','bekliyor','pasif')),
  dil text not null default 'tr' check (dil in ('tr','en','fa','ur')),
  departman_id bigint references departmanlar on delete set null,
  olusturma timestamptz default now()
);

create table projeler (
  id bigint generated always as identity primary key,
  departman_id bigint not null references departmanlar,
  ad text not null,
  musteri text,
  adres text,
  aciklama text,
  takipci_id uuid references profiller on delete set null,
  durum text not null default 'acik'
    check (durum in ('acik','devam','bitti','patronda','onaylandi','iade')),
  para_birimi text not null default 'TRY',   -- TRY, EUR, USD, GBP ... (ISO kodu)
  iscilik numeric default 0,
  fiyat numeric,
  fiyat_notu text,
  patron_notu text,
  baslangic date,
  bitis date,
  olusturan uuid references profiller on delete set null default auth.uid(),
  olusturma timestamptz default now(),
  gonderim timestamptz,
  onay_tarihi timestamptz,
  fatura_durum text check (fatura_durum in ('bekliyor','kesildi')),
  proforma_no text,
  proforma jsonb,                         -- Ofis'in düzenlediği proforma içeriği
  fatura_no text,
  fatura_tarihi timestamptz,
  odeme_durum text check (odeme_durum in ('bekliyor','kismi','alindi')),  -- fatura kesilince 'bekliyor'
  odenen numeric not null default 0,
  odeme_tarihi timestamptz                -- tamamı alınınca: iş arşive kalkar
);

create table gorevler (
  id bigint generated always as identity primary key,
  proje_id bigint not null references projeler on delete cascade,
  baslik text not null,
  aciklama text,
  atanan_id uuid references profiller on delete set null,          -- ilk kişi
  ekip uuid[] not null default '{}',                               -- işe atanan herkes (1-3 kişi)
  plan_tarih date,
  durum text not null default 'bekliyor'
    check (durum in ('bekliyor','bakiyor','basladi','bitti')),   -- basladi = Devam ediyor
  bitis_tarihi timestamptz,                                       -- aylık rapor için
  olusturma timestamptz default now()
);

create table islemler (
  id bigint generated always as identity primary key,
  proje_id bigint not null references projeler on delete cascade,
  gorev_id bigint references gorevler on delete set null,
  kullanici_id uuid references profiller on delete set null default auth.uid(),
  aciklama text not null,
  sure_saat numeric,
  tarih timestamptz default now()
);

create table malzemeler (
  id bigint generated always as identity primary key,
  proje_id bigint not null references projeler on delete cascade,
  tur text not null check (tur in ('alinan','kullanilan')),
  ad text not null,
  miktar numeric not null default 1,
  birim text default 'adet',
  birim_fiyat numeric not null default 0,
  para_birimi text not null default 'TRY',    -- malzemenin alındığı döviz
  giren uuid references profiller on delete set null default auth.uid(),
  fis_yolu text,                       -- fotoğrafı çekilen fiş (Storage: fisler)
  tarih timestamptz default now()
);

-- ---------- Yardımcı fonksiyonlar ----------
create function benim_rol() returns text
language sql stable security definer set search_path = public as $$
  select rol from profiller where id = auth.uid()
$$;

create function benim_dep() returns bigint
language sql stable security definer set search_path = public as $$
  select departman_id from profiller where id = auth.uid()
$$;

create function aktif() returns boolean
language sql stable security definer set search_path = public as $$
  select coalesce(benim_rol() in ('patron','ds','calisan'), false)
$$;

create function ofisten() returns boolean
language sql stable security definer set search_path = public as $$
  select coalesce((select d.ofis from departmanlar d where d.id = benim_dep()), false)
$$;

create function projeye_erisim(p bigint) returns boolean
language sql stable security definer set search_path = public as $$
  select aktif() and exists (
    select 1 from projeler pr where pr.id = p and (
      benim_rol() = 'patron'
      or (ofisten() and (pr.durum = 'onaylandi' or (pr.durum = 'patronda' and pr.proforma_no is not null)))
      or (benim_rol() = 'ds' and pr.departman_id = benim_dep())
      or pr.takipci_id = auth.uid()
      or exists (select 1 from gorevler g where g.proje_id = pr.id and (g.atanan_id = auth.uid() or auth.uid() = any(g.ekip)))
    )
  )
$$;

create function projeyi_yonetir(p bigint) returns boolean
language sql stable security definer set search_path = public as $$
  select exists (
    select 1 from projeler pr where pr.id = p and (
      benim_rol() = 'patron'
      or (benim_rol() = 'ds' and pr.departman_id = benim_dep())
    )
  )
$$;

-- Projenin tamamını görebilen: patron, departman sorumlusu, projeyi takip eden
create function projeyi_takip_eder(p bigint) returns boolean
language sql stable security definer set search_path = public as $$
  select aktif() and exists (
    select 1 from projeler pr where pr.id = p and (
      benim_rol() = 'patron'
      or (benim_rol() = 'ds' and pr.departman_id = benim_dep())
      or pr.takipci_id = auth.uid()
    )
  )
$$;

-- ---------- Yeni kayıt: ilk kişi İşveren, diğerleri onay bekler ----------
create function yeni_kullanici() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  insert into profiller (id, ad, kullanici_adi, rol)
  values (
    new.id,
    coalesce(nullif(new.raw_user_meta_data->>'ad',''), split_part(new.email,'@',1)),
    split_part(new.email,'@',1),
    case when not exists (select 1 from profiller) then 'patron' else 'bekliyor' end
  );
  return new;
end $$;

create trigger trg_yeni_kullanici
after insert on auth.users
for each row execute function yeni_kullanici();

-- ---------- Patrondaki iş kilidi + onay/iade sadece patron ----------
create function proje_kontrol() returns trigger
language plpgsql set search_path = public as $$
begin
  if current_setting('dotek.fatura', true) = '1' then return new; end if;
  if benim_rol() is distinct from 'patron' then
    if old.durum in ('patronda','onaylandi') then
      raise exception 'Bu iş işverende, değiştirilemez';
    end if;
    if new.durum in ('onaylandi','iade') and new.durum is distinct from old.durum then
      raise exception 'Onay ve iadeyi sadece işveren yapabilir';
    end if;
  end if;
  if new.durum = 'onaylandi' and old.durum is distinct from 'onaylandi' then
    new.fatura_durum := 'bekliyor';
  end if;
  return new;
end $$;

create trigger trg_proje_kontrol
before update on projeler
for each row execute function proje_kontrol();

-- ---------- Görev başlayınca proje "Devam ediyor" olur ----------
create function gorev_basladi() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  if new.durum in ('bakiyor','basladi') then
    update projeler set durum = 'devam' where id = new.proje_id and durum = 'acik';
  end if;
  return new;
end $$;

create trigger trg_gorev_basladi
after update of durum on gorevler
for each row execute function gorev_basladi();

-- Bir kişiye iş verilebilir mi? İşveren herkese, sorumlu sadece kendi departmanına
create function atanabilir(k uuid) returns boolean
language sql stable security definer set search_path = public as $$
  select k is null or benim_rol() = 'patron' or k = auth.uid()
      or exists (select 1 from profiller where id = k and departman_id = benim_dep())
$$;

create function ekip_atanabilir(e uuid[]) returns boolean
language sql stable security definer set search_path = public as $$
  select not exists (select 1 from unnest(coalesce(e, '{}')) k where not atanabilir(k))
$$;

-- Görevin ekibi her zaman ilk kişiyi de içerir
create function gorev_ekip_duzelt() returns trigger
language plpgsql set search_path = public as $$
begin
  if new.atanan_id is null and cardinality(new.ekip) > 0 then new.atanan_id := new.ekip[1]; end if;
  new.ekip := array(select distinct k from unnest(array_append(coalesce(new.ekip, '{}'), new.atanan_id)) k where k is not null);
  if new.durum = 'bitti' then
    if tg_op = 'INSERT' or old.durum is distinct from 'bitti' then new.bitis_tarihi := now(); end if;
  else
    new.bitis_tarihi := null;
  end if;
  return new;
end $$;
create trigger trg_gorev_ekip before insert or update on gorevler
for each row execute function gorev_ekip_duzelt();

-- ---------- Güvenlik kuralları (RLS) ----------
alter table departmanlar enable row level security;
alter table profiller   enable row level security;
alter table projeler    enable row level security;
alter table gorevler    enable row level security;
alter table islemler    enable row level security;
alter table malzemeler  enable row level security;

create policy dep_oku   on departmanlar for select to authenticated using (true);
create policy dep_yaz   on departmanlar for all    to authenticated
  using (benim_rol() = 'patron') with check (benim_rol() = 'patron');

create policy prf_oku   on profiller for select to authenticated
  using (aktif() or id = auth.uid());
create policy prf_yaz   on profiller for update to authenticated
  using (benim_rol() = 'patron') with check (benim_rol() = 'patron');

create policy prj_oku   on projeler for select to authenticated
  using (projeye_erisim(id));
create policy prj_ekle  on projeler for insert to authenticated
  with check ((benim_rol() = 'patron' or (benim_rol() = 'ds' and departman_id = benim_dep())) and atanabilir(takipci_id));
create policy prj_guncel on projeler for update to authenticated
  using (projeyi_yonetir(id))
  with check ((benim_rol() = 'patron' or (benim_rol() = 'ds' and departman_id = benim_dep())) and atanabilir(takipci_id));
create policy prj_sil   on projeler for delete to authenticated
  using (projeyi_yonetir(id));

-- Çalışan sadece kendisine verilen görevi görür
create policy grv_oku   on gorevler for select to authenticated
  using (projeyi_takip_eder(proje_id) or (aktif() and (atanan_id = auth.uid() or auth.uid() = any(ekip))));
create policy grv_ekle  on gorevler for insert to authenticated
  with check (projeyi_yonetir(proje_id) and atanabilir(atanan_id) and ekip_atanabilir(ekip));
create policy grv_guncel on gorevler for update to authenticated
  using (projeyi_yonetir(proje_id) or atanan_id = auth.uid() or auth.uid() = any(ekip))
  with check (atanan_id = auth.uid() or auth.uid() = any(ekip) or (atanabilir(atanan_id) and ekip_atanabilir(ekip)));
create policy grv_sil   on gorevler for delete to authenticated
  using (projeyi_yonetir(proje_id));

create policy isl_oku   on islemler for select to authenticated
  using (projeyi_takip_eder(proje_id) or (aktif() and kullanici_id = auth.uid()));
create policy isl_ekle  on islemler for insert to authenticated
  with check (projeye_erisim(proje_id) and kullanici_id = auth.uid());
create policy isl_sil   on islemler for delete to authenticated
  using (kullanici_id = auth.uid() or projeyi_yonetir(proje_id));

create policy mlz_oku   on malzemeler for select to authenticated
  using (projeyi_takip_eder(proje_id)
         or (ofisten() and exists (select 1 from projeler p where p.id = proje_id and p.durum = 'onaylandi')));
create policy mlz_yaz   on malzemeler for all to authenticated
  using (projeyi_yonetir(proje_id)) with check (projeyi_yonetir(proje_id));

-- ---------- Ofis: fatura kesildi ----------
create function fatura_kes(p bigint, no text) returns void
language plpgsql security definer set search_path = public as $$
begin
  if not (ofisten() or benim_rol() = 'patron') then
    raise exception 'Faturayı sadece Ofis işaretleyebilir';
  end if;
  perform set_config('dotek.fatura', '1', true);
  update projeler set fatura_durum = 'kesildi', fatura_no = nullif(trim(no), ''), fatura_tarihi = now(),
         odeme_durum = coalesce(odeme_durum, 'bekliyor')
   where id = p and durum = 'onaylandi';
  perform set_config('dotek.fatura', '', true);
  update bildirimler set islem_yapildi = true where proje_id = p and tur = 'fatura';
end $$;

-- ---------- Bildirimler ----------
create table bildirimler (
  id bigint generated always as identity primary key,
  alici_id uuid not null references profiller on delete cascade,
  metin text not null,
  proje_id bigint references projeler on delete cascade,
  departman_id bigint,          -- mesaj bildirimi için kanal (0 = Genel)
  tur text,
  islem_yapildi boolean not null default false,
  tarih timestamptz default now()
);
create index bildirim_alici on bildirimler (alici_id, islem_yapildi);
alter table bildirimler enable row level security;
create policy bld_oku    on bildirimler for select to authenticated using (alici_id = auth.uid());
create policy bld_guncel on bildirimler for update to authenticated
  using (alici_id = auth.uid()) with check (alici_id = auth.uid());

create function bildir(alici uuid, m text, p bigint, d bigint, t text) returns void
language sql security definer set search_path = public as $$
  insert into bildirimler (alici_id, metin, proje_id, departman_id, tur)
  select alici, m, p, d, t
  where alici is not null and alici is distinct from auth.uid()
$$;
revoke execute on function bildir(uuid,text,bigint,bigint,text) from public, anon, authenticated;

-- Görev verilince çalışana, görev bitince sorumluya
create function bld_gorev() returns trigger
language plpgsql security definer set search_path = public as $$
declare pr projeler; r record; k uuid; isimler text;
begin
  select * into pr from projeler where id = new.proje_id;
  select string_agg(ad, ', ') into isimler from profiller where id = any(new.ekip);
  -- Planlanınca işe atanan herkese bildirim
  foreach k in array new.ekip loop
    if tg_op = 'INSERT' or not (k = any(old.ekip)) then
      perform bildir(k, 'Sana yeni iş verildi: ' || new.baslik || ' · ' || pr.ad
        || case when cardinality(new.ekip) > 1 then ' · Ekip: ' || isimler else '' end, pr.id, null, 'gorev');
    end if;
  end loop;
  -- Ekip işe başlayınca "yeni iş" bildirimleri kapanır
  if tg_op = 'UPDATE' and old.durum = 'bekliyor' and new.durum <> 'bekliyor' then
    foreach k in array new.ekip loop
      if not exists (select 1 from gorevler where proje_id = new.proje_id and durum = 'bekliyor'
                     and (atanan_id = k or k = any(ekip))) then
        update bildirimler set islem_yapildi = true where alici_id = k and proje_id = new.proje_id and tur = 'gorev';
      end if;
    end loop;
  end if;
  if tg_op = 'UPDATE' and new.durum = 'bitti' and old.durum <> 'bitti' then
    for r in select id from profiller where (rol = 'ds' and departman_id = pr.departman_id) or id = pr.takipci_id loop
      perform bildir(r.id, coalesce(isimler, '') || ' bitirdi: ' || new.baslik || ' · ' || pr.ad, pr.id, null, 'bitti');
    end loop;
  end if;
  return new;
end $$;
create trigger trg_bld_gorev after insert or update on gorevler
for each row execute function bld_gorev();

-- İşverene gönderilince patrona, onay/iadede sorumluya
create function bld_proje() returns trigger
language plpgsql security definer set search_path = public as $$
declare r record;
begin
  if tg_op = 'INSERT' then
    perform bildir(new.takipci_id, 'Bir projenin takibi sana verildi: ' || new.ad, new.id, null, 'takip');
    return new;
  end if;
  if new.takipci_id is distinct from old.takipci_id then
    perform bildir(new.takipci_id, 'Bir projenin takibi sana verildi: ' || new.ad, new.id, null, 'takip');
  end if;
  if new.durum is distinct from old.durum then
    -- İşlem yapılınca ilgili bildirim kapanır
    if old.durum = 'patronda' then
      update bildirimler set islem_yapildi = true where proje_id = new.id and tur = 'onay';
    end if;
    if new.durum = 'patronda' then
      update bildirimler set islem_yapildi = true where proje_id = new.id and tur = 'iade';
    end if;
    if old.durum = 'bitti' then
      update bildirimler set islem_yapildi = true where proje_id = new.id and tur = 'is_bitti';
    end if;
    if new.durum = 'bitti' then
      -- Sorumlu işi bitirdi: işverene bilgi (proforma hazırlanacak)
      for r in select id from profiller where rol = 'patron' loop
        perform bildir(r.id, coalesce((select ad from profiller where id = new.takipci_id), (select ad from profiller where id = auth.uid()), 'Sorumlu') || ' işi bitirdi: ' || new.ad
          || coalesce(' · ' || new.musteri, ''), new.id, null, 'is_bitti');
      end loop;
    end if;
    if new.durum = 'patronda' then
      for r in select id from profiller where rol = 'patron' loop
        perform bildir(r.id, coalesce((select ad from profiller where id = new.takipci_id), 'Sorumlu') || ' faturayı hazırladı: ' || new.ad || ' · onayın bekleniyor', new.id, null, 'onay');
      end loop;
    elsif new.durum in ('onaylandi', 'iade') then
      for r in select id from profiller where (rol = 'ds' and departman_id = new.departman_id) or id = new.takipci_id loop
        perform bildir(r.id,
          case when new.durum = 'onaylandi' then 'İşveren onayladı: ' else 'İşveren iade etti: ' end
          || new.ad || coalesce(' — ' || new.patron_notu, ''), new.id, null, new.durum);
      end loop;
      if new.durum = 'onaylandi' then
        for r in select p.id from profiller p join departmanlar d on d.id = p.departman_id
                 where d.ofis and p.rol in ('ds','calisan') loop
          perform bildir(r.id, 'Fatura için bekliyor: ' || new.ad || ' · ' || coalesce(new.fiyat, 0)::text || ' ' || new.para_birimi, new.id, null, 'fatura');
        end loop;
      end if;
    end if;
  end if;
  if new.fatura_durum = 'kesildi' and old.fatura_durum is distinct from 'kesildi' then
    for r in select id from profiller where rol = 'patron' or (rol = 'ds' and departman_id = new.departman_id) loop
      perform bildir(r.id, case when new.iletim = 'proforma' then 'Proforma müşteriye verildi: ' else 'Fatura kesildi: ' end || new.ad || ' · ' || coalesce(new.fiyat, 0)::text || ' ' || new.para_birimi || ' tahsil edilecek' || coalesce(' · No: ' || new.fatura_no, ''), new.id, null, 'kesildi');
    end loop;
  end if;
  return new;
end $$;
create trigger trg_bld_proje after insert or update on projeler
for each row execute function bld_proje();

-- Mesaj gelince departmana (Genel ise herkese)
create function bld_mesaj() returns trigger
language plpgsql security definer set search_path = public as $$
declare r record; kim text; kanal text;
begin
  select ad into kim from profiller where id = new.gonderen_id;
  select ad into kanal from departmanlar where id = new.departman_id;
  -- Cevap yazan kişinin o kanaldaki mesaj bildirimleri kapanır
  update bildirimler set islem_yapildi = true
   where alici_id = new.gonderen_id and tur = 'mesaj' and departman_id = coalesce(new.departman_id, 0);
  -- Alıcılar: kanalın departmanı + o kanalda daha önce yazmış başka departmandakiler (cevap onlara da gider).
  -- İşverene sadece Genel kanal ve kendi yazıştığı kanallar gider.
  for r in select id from profiller
           where rol in ('patron','ds','calisan')
             and (new.departman_id is null or departman_id = new.departman_id
                  or id in (select gonderen_id from mesajlar m
                             where coalesce(m.departman_id, 0) = coalesce(new.departman_id, 0)
                               and m.id <> new.id and m.tarih > now() - interval '30 days')) loop
    perform bildir(r.id, coalesce(kim, '') || ' mesaj yazdı (' || coalesce(kanal, 'Genel') || '): ' || left(new.metin, 80),
                   null, coalesce(new.departman_id, 0), 'mesaj');
  end loop;
  return new;
end $$;
create trigger trg_bld_mesaj after insert on mesajlar
for each row execute function bld_mesaj();

-- Yeni personel kayıt olunca patrona
create function bld_kayit() returns trigger
language plpgsql security definer set search_path = public as $$
declare r record;
begin
  if new.rol = 'bekliyor' then
    for r in select id from profiller where rol = 'patron' loop
      perform bildir(r.id, 'Yeni personel onay bekliyor: ' || new.ad, null, null, 'kayit');
    end loop;
  end if;
  return new;
end $$;
create trigger trg_bld_kayit after insert on profiller
for each row execute function bld_kayit();

-- İşveren rol verince "yeni personel" bildirimi kapanır
create function bld_kayit_kapat() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  if old.rol = 'bekliyor' and new.rol <> 'bekliyor' then
    update bildirimler set islem_yapildi = true
     where tur = 'kayit' and metin = 'Yeni personel onay bekliyor: ' || new.ad;
  end if;
  return new;
end $$;
create trigger trg_bld_kayit_kapat after update on profiller
for each row execute function bld_kayit_kapat();

-- ---------- Ajanda: patronun kendi planları ve dağıttığı işler ----------
create table ajanda (
  id bigint generated always as identity primary key,
  sahip_id uuid references profiller on delete cascade default auth.uid(),   -- planı yazan
  atanan_id uuid references profiller on delete set null,                   -- yapacak kişi (kendisi olabilir)
  baslik text not null,
  musteri text,                                                             -- kimin işi (proje açılınca projeye geçer)
  aciklama text,
  tarih date,

-- Ad + ayrılma eki: Doğuş → Doğuş'tan, Tolga → Tolga'dan, İbrahim → İbrahim'den (ilk isim kullanılır)
create function ayrilma_eki(ad text) returns text language plpgsql immutable as $$
declare a text := split_part(coalesce(nullif(trim(ad), ''), 'Biri'), ' ', 1);
  son text := lower(right(a, 1)); sesli text; ek text;
begin
  sesli := lower(right(regexp_replace(a, '[^aeıioöuüAEIİOÖUÜ]', '', 'g'), 1));
  ek := case when son = any (array['f','s','t','k','ç','ş','h','p']) then 't' else 'd' end
        || case when sesli = any (array['e','i','ö','ü']) then 'en' else 'an' end;
  return a || '''' || ek;
end $$;

create function bld_ajanda() returns trigger
language plpgsql security definer set search_path = public as $$
declare kim text;
begin
  select ad into kim from profiller where id = new.sahip_id;
  if (tg_op = 'INSERT' or new.atanan_id is distinct from old.atanan_id)
     and new.atanan_id is not null and new.atanan_id is distinct from auth.uid() then
    insert into bildirimler (alici_id, metin, tur, ajanda_id)
    values (new.atanan_id, ayrilma_eki(kim) || ' iş geldi: ' || new.baslik
            || coalesce(' · ' || to_char(new.tarih, 'DD.MM.YYYY'), ''), 'ajanda', new.id);
  end if;
  if tg_op = 'UPDATE' and old.durum = 'bekliyor' and new.durum <> 'bekliyor' then
    -- kişi işe bakmaya başladı: "iş verildi" bildirimi kapanır
    update bildirimler set islem_yapildi = true where ajanda_id = new.id and tur = 'ajanda';
  end if;
  if tg_op = 'UPDATE' and new.durum = 'planlandi' and old.durum <> 'planlandi' and new.sahip_id is distinct from auth.uid() then
    insert into bildirimler (alici_id, metin, tur, ajanda_id)
    values (new.sahip_id, coalesce((select ad from profiller where id = new.atanan_id), '') || ' planladı: ' || new.baslik
            || ' · ekibine dağıttı', 'ajanda_plan', new.id);
  end if;
  if tg_op = 'UPDATE' and new.durum = 'bitti' and old.durum <> 'bitti' then
    update bildirimler set islem_yapildi = true where ajanda_id = new.id and tur = 'ajanda';
    if new.sahip_id is distinct from new.atanan_id and new.sahip_id is distinct from auth.uid() then
      insert into bildirimler (alici_id, metin, tur, ajanda_id)
      values (new.sahip_id, coalesce((select ad from profiller where id = new.atanan_id), '') || ' bitirdi: ' || new.baslik
              || coalesce(' · Ücret: ' || new.ucret::text || ' ' || new.para_birimi, ''), 'ajanda_bitti', new.id);
    end if;
  end if;
  return new;
end $$;
create trigger trg_bld_ajanda after insert or update on ajanda
for each row execute function bld_ajanda();

-- Kabul edilen teklif → departmanda yeni proje
create function teklif_projeye(t bigint, dep bigint) returns bigint
language plpgsql security definer set search_path = public as $$
declare x teklifler; pid bigint; r record;
begin
  select * into x from teklifler where id = t;
  if not (ofisten() or benim_rol() = 'patron'
          or (benim_rol() = 'ds' and x.olusturan = auth.uid() and dep = benim_dep())) then
    raise exception 'Bu teklifi projeye çeviremezsin';
  end if;
  if x.proje_id is not null then return x.proje_id; end if;
  insert into projeler (departman_id, ad, musteri, adres, aciklama, para_birimi, takipci_id)
  values (dep, coalesce(nullif(x.proje_adi, ''), nullif(x.konu, ''), x.musteri), x.musteri, x.adres,
          'Teklif ' || x.no || ' kabul edildi.' || coalesce(' Konu: ' || x.konu, ''), x.para_birimi,
          (select id from profiller where rol = 'ds' and departman_id = dep limit 1))
  returning id into pid;
  update teklifler set durum = 'kabul', proje_id = pid, departman_id = dep where id = t;
  for r in select id from profiller where rol = 'ds' and departman_id = dep loop
    perform bildir(r.id, 'Yeni proje (teklif kabul edildi): ' || x.musteri || coalesce(' · ' || x.konu, ''), pid, null, 'takip');
  end loop;
  for r in select id from profiller where rol = 'patron' or id = x.olusturan loop
    perform bildir(r.id, 'Teklif kabul edildi: ' || x.musteri || ' · ' || x.no || ' → ' || coalesce((select ad from departmanlar where id = dep), '')
      || ' departmanında proje açıldı', pid, null, 'teklif_kabul');
  end loop;
  return pid;
end $$;

-- ---------- Teklif isteği bildirimleri ----------
create function bld_teklif() returns trigger
language plpgsql security definer set search_path = public as $$
declare r record; kim text;
begin
  select ad into kim from profiller where id = auth.uid();
  if tg_op = 'INSERT' and new.durum = 'istendi' then
    for r in select p.id from profiller p join departmanlar d on d.id = p.departman_id
             where d.ofis and p.rol in ('ds','calisan') loop
      insert into bildirimler (alici_id, metin, tur) values (r.id, coalesce(kim, '') || ' teklif istedi: ' || new.musteri || coalesce(' · ' || left(new.konu, 60), ''), 'teklif');
    end loop;
  elsif tg_op = 'UPDATE' and new.durum = 'red' and old.durum <> 'red' then
    for r in select id from profiller where (rol = 'patron' or id = new.olusturan) and id is distinct from auth.uid() loop
      insert into bildirimler (alici_id, metin, tur) values (r.id, 'Teklif reddedildi: ' || new.musteri || ' · ' || new.no, 'teklif_red');
    end loop;
  elsif tg_op = 'UPDATE' and new.durum = 'istendi' and old.acilma_tarih is null and new.acilma_tarih is not null then
    -- Ofis isteği açtı: isteyene "inceliyor" bilgisi
    for r in select id from profiller where id = new.olusturan and id is distinct from auth.uid() loop
      insert into bildirimler (alici_id, metin, tur) values (r.id, coalesce(kim, 'Ofis') || ' teklif isteğini açtı, inceliyor: ' || new.musteri, 'teklif_acildi');
    end loop;
  elsif tg_op = 'UPDATE' and old.durum = 'istendi' and new.durum <> 'istendi' then
    update bildirimler set islem_yapildi = true where tur = 'teklif' and metin like '%teklif istedi: ' || new.musteri || '%';
    for r in select id from profiller where id = new.olusturan and id is distinct from auth.uid() loop
      insert into bildirimler (alici_id, metin, tur) values (r.id, coalesce(kim, 'Ofis') || ' teklifi hazırlıyor: ' || new.musteri || ' · ' || new.no, 'teklif_hazir');
    end loop;
  end if;
  if tg_op = 'UPDATE' and new.durum = 'gonderildi' and old.durum is distinct from 'gonderildi' then
    for r in select id from profiller where (rol = 'patron' or id = new.olusturan) and id is distinct from auth.uid() loop
      insert into bildirimler (alici_id, metin, tur) values (r.id, 'Teklif müşteriye gönderildi: ' || new.musteri || ' · ' || new.no
        || coalesce(' · ' || new.gonderim_kanal, ''), 'teklif_gonderildi');
    end loop;
  end if;
  return new;
end $$;
create trigger trg_bld_teklif after insert or update on teklifler
for each row execute function bld_teklif();

-- ---------- İşveren departman sorumlusuna iş verince proje otomatik açılır ----------
-- (Ofis ve çalışanlara verilen işler ajanda olarak kalır.) Sorumlu ilk görevi atayınca iş "Planlandı" olur.
create function ajanda_proje_ac() returns trigger
language plpgsql security definer set search_path = public as $$
declare k profiller; d departmanlar; pid bigint;
begin
  if new.proje_id is not null or new.atanan_id is null or new.atanan_id = new.sahip_id then return new; end if;
  select * into k from profiller where id = new.atanan_id;
  if k.rol is distinct from 'ds' then return new; end if;
  select * into d from departmanlar where id = k.departman_id;
  if d.id is null or coalesce(d.ofis, false) then return new; end if;
  insert into projeler (departman_id, ad, musteri, aciklama, takipci_id, baslangic, olusturan)
  values (d.id, new.baslik, nullif(trim(new.musteri), ''),
          coalesce(new.aciklama, (select ad from profiller where id = new.sahip_id) || ' tarafından verildi'),
          k.id, new.tarih, new.sahip_id)
  returning id into pid;
  -- "takip sana verildi" bildirimi gereksiz: "iş geldi" bildirimi zaten gidiyor
  delete from bildirimler where proje_id = pid and tur = 'takip';
  new.proje_id := pid;
  return new;
end $$;
create trigger trg_ajanda_proje_ac before insert on ajanda
for each row execute function ajanda_proje_ac();

-- ---------- Tahsilat: faturası kesilen işin ödemesi ----------

create table tahsilatlar (
  id bigint generated always as identity primary key,
  proje_id bigint not null references projeler on delete cascade,
  tutar numeric not null check (tutar > 0),
  para_birimi text not null default 'TRY',
  yontem text not null default 'havale' check (yontem in ('nakit','havale','kart','cek')),

  olusturma timestamptz default now()
);
alter table tahsilatlar enable row level security;
create policy ths_oku  on tahsilatlar for select to authenticated using (ofisten() or projeye_erisim(proje_id));
create policy ths_ekle on tahsilatlar for insert to authenticated
  with check ((ofisten() or benim_rol() = 'patron') and giren = auth.uid()
              and exists (select 1 from projeler p where p.id = proje_id and p.fatura_durum = 'kesildi'));
create policy ths_sil  on tahsilatlar for delete to authenticated using (ofisten() or benim_rol() = 'patron');

create function tahsilat_guncelle() returns trigger
language plpgsql security definer set search_path = public as $$
declare pid bigint := coalesce(new.proje_id, old.proje_id); pr projeler; top numeric; yeni text; r record;
begin
  select * into pr from projeler where id = pid;
  select coalesce(sum(tutar), 0) into top from tahsilatlar where proje_id = pid;
  yeni := case when top <= 0 then 'bekliyor' when top >= coalesce(pr.fiyat, 0) then 'alindi' else 'kismi' end;
  perform set_config('dotek.fatura', '1', true);
  update projeler set odenen = top, odeme_durum = yeni,
         odeme_tarihi = case when yeni = 'alindi' then coalesce(odeme_tarihi, now()) end
   where id = pid;
  perform set_config('dotek.fatura', '', true);
  if tg_op = 'INSERT' then
    for r in select id from profiller where rol = 'patron' or (rol = 'ds' and departman_id = pr.departman_id) loop
      perform bildir(r.id, case when yeni = 'alindi' then 'Ödeme alındı, iş arşive kalktı: ' else 'Kısmi ödeme alındı: ' end
        || pr.ad || ' · ' || new.tutar::text || ' ' || new.para_birimi, pid, null, 'odeme');
    end loop;
  end if;
  return null;
end $$;
create trigger trg_tahsilat after insert or delete on tahsilatlar
for each row execute function tahsilat_guncelle();

-- ---------- Proforma sahadan: sorumlu (veya işveren) biten işin proformasını hazırlar ve gönderir ----------
-- Sorumlu gönderince iş işverenin onayına düşer, Ofis de görür. İşveren kendisi gönderirse onaylanmış sayılır.
-- Fatura kesilene kadar düzeltilebilir; düzeltmede işverene ve Ofis'e "güncellendi" bildirimi gider.
create function proforma_gonder(p bigint, veri jsonb, gonder boolean, toplam numeric, isc numeric) returns text
language plpgsql security definer set search_path = public as $$
declare pr projeler; no text; kim text; r record; ptr boolean := benim_rol() = 'patron';
begin
  select * into pr from projeler where id = p;
  if not found then raise exception 'Proje bulunamadı'; end if;
  if not (ptr or (benim_rol() = 'ds' and pr.departman_id = benim_dep())) then
    raise exception 'Bu işin proformasını hazırlayamazsın';
  end if;
  if pr.fatura_durum = 'kesildi' then raise exception 'Fatura kesildi, proforma artık değiştirilemez'; end if;
  no := coalesce(pr.proforma_no, 'PRF-' || to_char(now(), 'YYYY') || '-' || lpad(nextval('proforma_sira')::text, 4, '0'));
  select ad into kim from profiller where id = auth.uid();
  perform set_config('dotek.fatura', '1', true);
  update projeler set proforma_no = no, proforma = jsonb_set(veri, '{no}', to_jsonb(no)), fiyat = toplam, iscilik = coalesce(isc, 0),
         para_birimi = coalesce(nullif(veri->>'pb', ''), para_birimi)
   where id = p;
  if gonder then
    if pr.durum in ('acik', 'devam', 'bitti', 'iade') then
      if ptr then
        update projeler set durum = 'onaylandi', onay_tarihi = now(), gonderim = coalesce(gonderim, now()),
               fatura_durum = 'bekliyor', patron_notu = null where id = p;
      else
        update projeler set durum = 'patronda', gonderim = now(), patron_notu = null where id = p;
        for r in select pf.id from profiller pf join departmanlar d on d.id = pf.departman_id
                 where d.ofis and pf.rol in ('ds', 'calisan') loop
          perform bildir(r.id, coalesce(kim, '') || ' proforma hazırladı: ' || pr.ad || ' · ' || no || ' · İşveren onayı bekleniyor',
                         p, null, 'proforma');
        end loop;
      end if;
    else
      for r in select pf.id from profiller pf left join departmanlar d on d.id = pf.departman_id
               where pf.rol = 'patron' or (coalesce(d.ofis, false) and pf.rol in ('ds', 'calisan')) loop
        perform bildir(r.id, coalesce(kim, '') || ' proformayı güncelledi: ' || pr.ad || ' · ' || no || ' · '
                       || toplam::text || ' ' || coalesce(pr.para_birimi, 'TRY'), p, null, 'proforma');
      end loop;
    end if;
  end if;
  perform set_config('dotek.fatura', '', true);
  return no;
end $$;

-- Bitmiş görünen işe yeni görev eklenir ya da bir görev yeniden başlarsa iş "Devam ediyor"a döner
create or replace function gorev_basladi() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  if new.durum in ('bakiyor', 'basladi') or tg_op = 'INSERT' then
    perform set_config('dotek.fatura', '1', true);
    update projeler set durum = 'devam' where id = new.proje_id
       and (durum = 'bitti' or (durum = 'acik' and new.durum in ('bakiyor', 'basladi')));
    perform set_config('dotek.fatura', '', true);
  end if;
  return new;
end $$;
create trigger trg_gorev_eklendi after insert on gorevler for each row execute function gorev_basladi();
alter table malzemeler add column if not exists tedarikci text;   -- kimden alındı (firma); sadece içeride görünür, proformaya çıkmaz

-- Son görev de bitince iş otomatik "İş bitti" olur → işverene "proforma hazırlanacak" bildirimi gider
create or replace function gorev_hepsi_bitti() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  if new.durum = 'bitti' and old.durum is distinct from 'bitti'
     and not exists (select 1 from gorevler where proje_id = new.proje_id and durum <> 'bitti') then
    perform set_config('dotek.fatura', '1', true);
    update projeler set durum = 'bitti' where id = new.proje_id and durum in ('acik', 'devam');
    perform set_config('dotek.fatura', '', true);
  end if;
  return new;
end $$;
drop trigger if exists trg_gorev_hepsi_bitti on gorevler;
create trigger trg_gorev_hepsi_bitti after update of durum on gorevler for each row execute function gorev_hepsi_bitti();

-- Ofis müşteriye proforma da verebilir: "Proforma verildi" ya da "Fatura kesildi" → müşteriye iletildi, ödeme beklenir
alter table projeler add column if not exists iletim text check (iletim in ('proforma', 'fatura'));
drop function if exists fatura_kes(bigint, text);
create function fatura_kes(p bigint, no text, belge text default 'fatura') returns void
language plpgsql security definer set search_path = public as $$
begin
  if not (ofisten() or benim_rol() = 'patron') then
    raise exception 'Faturayı sadece Ofis işaretleyebilir';
  end if;
  perform set_config('dotek.fatura', '1', true);
  update projeler set iletim = case when belge = 'proforma' then 'proforma' else 'fatura' end,
         fatura_durum = 'kesildi', fatura_no = nullif(trim(no), ''), fatura_tarihi = now(),
         odeme_durum = coalesce(odeme_durum, 'bekliyor')
   where id = p and durum = 'onaylandi';
  perform set_config('dotek.fatura', '', true);
  update bildirimler set islem_yapildi = true where proje_id = p and tur = 'fatura';
end $$;

-- ---------- Ödeme takibi notları (Ofis / Doğuş / sorumlu: "aradım, cevap vermedi" gibi) ----------
create table if not exists tahsilat_notlar (
  id bigint generated always as identity primary key,
  proje_id bigint not null references projeler on delete cascade,
  yazan_id uuid references profiller on delete set null,
  metin text not null,
  bildirildi boolean not null default false,
  tarih timestamptz default now()
);
create index if not exists tahsilat_not_proje on tahsilat_notlar (proje_id, tarih);
alter table tahsilat_notlar enable row level security;
drop policy if exists tn_oku on tahsilat_notlar;
create policy tn_oku on tahsilat_notlar for select to authenticated
  using (benim_rol() = 'patron' or ofisten()
         or exists (select 1 from projeler p join profiller m on m.id = auth.uid()
                     where p.id = proje_id and m.rol = 'ds' and m.departman_id = p.departman_id));

create or replace function tahsilat_not(p bigint, metin text, bildir boolean default false) returns void
language plpgsql security definer set search_path = public as $$
declare pr projeler; kim text; r record;
begin
  select * into pr from projeler where id = p;
  if pr.id is null then raise exception 'İş bulunamadı'; end if;
  if not (benim_rol() = 'patron' or ofisten()
          or exists (select 1 from profiller where id = auth.uid() and rol = 'ds' and departman_id = pr.departman_id)) then
    raise exception 'Bu işe not yazamazsın';
  end if;
  if coalesce(trim(metin), '') = '' then raise exception 'Açıklama boş olamaz'; end if;
  insert into tahsilat_notlar (proje_id, yazan_id, metin, bildirildi) values (p, auth.uid(), trim(metin), coalesce(bildir, false));
  update bildirimler set islem_yapildi = true where alici_id = auth.uid() and proje_id = p and tur = 'tahsilat';
  if bildir then
    select ad into kim from profiller where id = auth.uid();
    for r in select id from profiller
              where case when benim_rol() = 'patron'
                         then (rol <> 'calisan' and departman_id in (select id from departmanlar where ofis)) or id = pr.takipci_id
                         else rol = 'patron' end loop
      perform bildir(r.id, coalesce(kim, '') || ' · ' || pr.ad || coalesce(' (' || pr.musteri || ')', '') || ': ' || trim(metin), p, null, 'tahsilat');
    end loop;
  end if;
end $$;
grant execute on function tahsilat_not(bigint, text, boolean) to authenticated;

-- Ödeme tamamen alınınca ödeme takibi bildirimleri kapanır
create or replace function tahsilat_bld_kapat() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  if new.odeme_durum = 'alindi' and old.odeme_durum is distinct from 'alindi' then
    update bildirimler set islem_yapildi = true where proje_id = new.id and tur = 'tahsilat';
  end if;
  return new;
end $$;
drop trigger if exists trg_tahsilat_bld_kapat on projeler;
create trigger trg_tahsilat_bld_kapat after update of odeme_durum on projeler for each row execute function tahsilat_bld_kapat();

-- Ofis (Gökçe) ana sayfasında bütün projeleri ve kimin hangi işte olduğunu görür (sadece okur, değiştiremez)
create or replace function projeye_erisim(p bigint) returns boolean
language sql stable security definer set search_path = public as $$
  select aktif() and exists (
    select 1 from projeler pr where pr.id = p and (
      benim_rol() = 'patron'
      or ofisten()
      or (benim_rol() = 'ds' and pr.departman_id = benim_dep())
      or pr.takipci_id = auth.uid()
      or exists (select 1 from gorevler g where g.proje_id = pr.id and (g.atanan_id = auth.uid() or auth.uid() = any(g.ekip)))
    )
  )
$$;
drop policy if exists grv_oku_ofis on gorevler;
create policy grv_oku_ofis on gorevler for select to authenticated using (aktif() and ofisten());

-- Müşteriye sadece proforma verildiyse (resmi fatura kesilmediyse) ödeme alınana kadar proforma düzeltilebilir
create or replace function proforma_gonder(p bigint, veri jsonb, gonder boolean, toplam numeric, isc numeric) returns text
language plpgsql security definer set search_path = public as $$
declare pr projeler; no text; kim text; r record; ptr boolean := benim_rol() = 'patron';
begin
  select * into pr from projeler where id = p;
  if not found then raise exception 'Proje bulunamadı'; end if;
  if not (ptr or (benim_rol() = 'ds' and pr.departman_id = benim_dep())) then
    raise exception 'Bu işin proformasını hazırlayamazsın';
  end if;
  if pr.fatura_durum = 'kesildi' and (coalesce(pr.iletim, 'fatura') <> 'proforma' or pr.odeme_durum = 'alindi') then
    raise exception 'Fatura kesildi, proforma artık değiştirilemez';
  end if;
  no := coalesce(pr.proforma_no, 'PRF-' || to_char(now(), 'YYYY') || '-' || lpad(nextval('proforma_sira')::text, 4, '0'));
  select ad into kim from profiller where id = auth.uid();
  perform set_config('dotek.fatura', '1', true);
  update projeler set proforma_no = no, proforma = jsonb_set(veri, '{no}', to_jsonb(no)), fiyat = toplam, iscilik = coalesce(isc, 0),
         para_birimi = coalesce(nullif(veri->>'pb', ''), para_birimi)
   where id = p;
  if gonder then
    if pr.durum in ('acik', 'devam', 'bitti', 'iade') then
      if ptr then
        update projeler set durum = 'onaylandi', onay_tarihi = now(), gonderim = coalesce(gonderim, now()),
               fatura_durum = 'bekliyor', patron_notu = null where id = p;
      else
        update projeler set durum = 'patronda', gonderim = now(), patron_notu = null where id = p;
        for r in select pf.id from profiller pf join departmanlar d on d.id = pf.departman_id
                 where d.ofis and pf.rol in ('ds', 'calisan') loop
          perform bildir(r.id, coalesce(kim, '') || ' proforma hazırladı: ' || pr.ad || ' · ' || no || ' · İşveren onayı bekleniyor',
                         p, null, 'proforma');
        end loop;
      end if;
    else
      for r in select pf.id from profiller pf left join departmanlar d on d.id = pf.departman_id
               where pf.rol = 'patron' or (coalesce(d.ofis, false) and pf.rol in ('ds', 'calisan')) loop
        perform bildir(r.id, coalesce(kim, '') || ' proformayı güncelledi: ' || pr.ad || ' · ' || no || ' · '
                       || toplam::text || ' ' || coalesce(pr.para_birimi, 'TRY'), p, null, 'proforma');
      end loop;
    end if;
  end if;
  perform set_config('dotek.fatura', '', true);
  return no;
end $$;

-- Alımlarda firma adını düzelt / birleştir (malzeme kaydı, proformadaki "Kimden alındı" ve firmaya yapılan ödemeler)
create or replace function firma_ata(kalemler jsonb, firma text, eski text default null) returns void
language plpgsql security definer set search_path = public as $$
declare e jsonb; f text := nullif(trim(firma), ''); pid bigint;
begin
  perform set_config('dotek.fatura', '1', true);
  for e in select * from jsonb_array_elements(coalesce(kalemler, '[]'::jsonb)) loop
    if e ? 'm' then
      update malzemeler set tedarikci = f where id = (e->>'m')::bigint
         and (benim_rol() = 'patron' or ofisten() or projeyi_yonetir(proje_id));
    elsif e ? 'p' then
      pid := (e->>'p')::bigint;
      update projeler set proforma = jsonb_set(proforma, array['kalemler', e->>'i', 'firma'], to_jsonb(coalesce(f, '')))
       where id = pid and proforma ? 'kalemler' and jsonb_array_length(proforma->'kalemler') > (e->>'i')::int
         and (benim_rol() = 'patron' or ofisten() or (benim_rol() = 'ds' and departman_id = benim_dep()));
    end if;
  end loop;
  if eski is not null and f is not null and (benim_rol() = 'patron' or ofisten()) then
    update tedarik_odemeler set firma = f where lower(trim(tedarik_odemeler.firma)) = lower(trim(eski));
  end if;
  perform set_config('dotek.fatura', '', true);
end $$;
grant execute on function firma_ata(jsonb, text, text) to authenticated;
